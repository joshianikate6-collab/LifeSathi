// स्प्लैश स्क्रीन
class SplashScreen extends StatefulWidget {
  @override
  _SplashScreenState createState() => _SplashScreenState();
}

class _SplashScreenState extends State<SplashScreen> {
  @override
  void initState() {
    super.initState();
    Future.delayed(Duration(seconds: 2), () {
      Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => HomeScreen()));
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Container(
        decoration: BoxDecoration(
          gradient: LinearGradient(colors: [Colors.teal.shade300, Colors.teal.shade700]),
        ),
        child: Center(
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Icon(Icons.favorite, size: 100, color: Colors.white),
              SizedBox(height: 20),
              Text('LifeSathi', style: TextStyle(fontSize: 32, fontWeight: FontWeight.bold, color: Colors.white)),
              CircularProgressIndicator(color: Colors.white),
            ],
          ),
        ),
      ),
    );
  }
}

// Bills Screen
class BillsScreen extends StatefulWidget {
  @override
  _BillsScreenState createState() => _BillsScreenState();
}

class _BillsScreenState extends State<BillsScreen> {
  List<String> bills = [];

  @override
  void initState() {
    super.initState();
    loadBills();
  }

  void loadBills() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    setState(() {
      bills = prefs.getStringList('bills') ?? [];
    });
  }

  void saveBills() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    prefs.setStringList('bills', bills);
  }

  void addBill(String bill) {
    setState(() {
      bills.add(bill);
      saveBills();
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Bills')),
      body: Column(
        children: [
          Expanded(
            child: ListView.builder(
              itemCount: bills.length,
              itemBuilder: (context, index) => ListTile(
                title: Text(bills[index]),
                trailing: IconButton(icon: Icon(Icons.delete), onPressed: () {
                  setState(() {
                    bills.removeAt(index);
                    saveBills();
                  });
                }),
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(16.0),
            child: ElevatedButton(
              onPressed: () async {
                String? newBill = await showDialog<String>(
                  context: context,
                  builder: (context) => AlertDialog(
                    title: Text('Add Bill'),
                    content: TextField(onSubmitted: (value) => Navigator.pop(context, value)),
                  ),
                );
                if (newBill != null && newBill.isNotEmpty) addBill(newBill);
              },
              child: Text('Add Bill'),
            ),
          ),
        ],
      ),
    );
  }
}

// Health Screen
class HealthScreen extends StatefulWidget {
  @override
  _HealthScreenState createState() => _HealthScreenState();
}

class _HealthScreenState extends State<HealthScreen> {
  double weight = 70.0;
  int steps = 5000;

  @override
  void initState() {
    super.initState();
    loadHealth();
  }

  void loadHealth() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    setState(() {
      weight = prefs.getDouble('weight') ?? 70.0;
      steps = prefs.getInt('steps') ?? 5000;
    });
  }

  void saveHealth() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    prefs.setDouble('weight', weight);
    prefs.setInt('steps', steps);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Health')),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            Text('Weight: ${weight}kg', style: TextStyle(fontSize: 20)),
            Slider(value: weight, min: 40, max: 150, onChanged: (value) {
              setState(() {
                weight = value;
                saveHealth();
              });
            }),
            SizedBox(height: 20),
            Text('Steps: $steps', style: TextStyle(fontSize: 20)),
            Slider(value: steps.toDouble(), min: 0, max: 20000, onChanged: (value) {
              setState(() {
                steps = value.toInt();
                saveHealth();
              });
            }),
          ],
        ),
      ),
    );
  }
}

// Habits Screen
class HabitsScreen extends StatefulWidget {
  @override
  _HabitsScreenState createState() => _HabitsScreenState();
}

class _HabitsScreenState extends State<HabitsScreen> {
  List<Map<String, dynamic>> habits = [];

  @override
  void initState() {
    super.initState();
    loadHabits();
  }

  void loadHabits() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    List<String>? habitStrings = prefs.getStringList('habits');

      setState(() {
        habits = habitStrings.map((e) => Map<String, dynamic>.from(e as Map)).toList();

    }

    SharedPreferences prefs = await SharedPreferences.getInstance();
    prefs.setStringList('habits', habits.map((e) => e.toString()).toList());
  }

  void addHabit(String habit) {
    setState(() {
      habits.add({'name': habit, 'streak': 0});
      saveHabits();
    });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('Habits')),
      body: Column(
        children: [
          Expanded(
            child: ListView.builder(

              itemBuilder: (context, index) => ListTile(
                title: Text(habits[index]['name']),
                subtitle: Text('Streak: ${habits[index]['streak']}'),

// Money Tracking Screen
class MoneyTrackingScreen extends StatefulWidget {
  @override
  _MoneyTrackingScreenState createState() => _MoneyTrackingScreenState();
}

class _MoneyTrackingScreenState extends State<MoneyTrackingScreen> {
  List<Map<String, dynamic>> transactions = [];

  @override
  void initState() {
    super.initState();
    loadTransactions();
  }

  void loadTransactions() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    List<String>? transStrings = prefs.getStringList('transactions');
    if (transStrings != null) {
      setState(() {
        transactions = transStrings.map((e) => Map<String, dynamic>.from(e as Map)).toList();
      });
    }
  }

  void saveTransactions() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    prefs.setStringList('transactions', transactions.map((e) => e.toString()).toList());
  }

  void addTransaction(String desc, double amount, bool isIncome) {
    setState(() {
      transactions.add({'desc': desc, 'amount': amount, 'isIncome': isIncome});
      saveTransactions();
    });
  }

  @override
  Widget build(BuildContext context) {
    double totalIncome = transactions.where((t) => t['isIncome']).fold(0.0, (sum, t) => sum + t['amount']);
    double totalExpense = transactions.where((t) => !t['isIncome']).fold(0.0, (sum, t) => sum + t['amount']);

    return Scaffold(
      appBar: AppBar(title: Text('Money Tracking')),
      body: Column(
        children: [
          Text('Total Income: $totalIncome | Expense: $totalExpense'),
          SizedBox(height: 20),
          Container(
            height: 200,
            child: charts.BarChart(
              [
                charts.Series<Map<String, dynamic>, String>(
                  id: 'Money',
                  data: [{'type': 'Income', 'amount': totalIncome}, {'type': 'Expense', 'amount': totalExpense}],
                  domainFn: (d, _) => d['type'],
                  measureFn: (d, _) => d['amount'],
                ),
              ],
            ),
          ),
          Expanded(
            child: ListView.builder(
              itemCount: transactions.length,
              itemBuilder: (context, index) => ListTile(
                title: Text(transactions[index]['desc']),
                subtitle: Text('${transactions[index]['isIncome'] ? '+' : '-'}${transactions[index]['amount']}'),
              ),
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(16.0),
            child: ElevatedButton(
              onPressed: () async {
                String? desc = await showDialog<String>(
                  context: context,
                  builder: (context) => AlertDialog(
                    title: Text('Add Transaction'),
                    content: TextField(onSubmitted: (value) => Navigator.pop(context, value)),
                  ),
                );
                if (desc != null) addTransaction(desc, 100.0, true);
              },
              child: Text('Add Transaction'),
            ),
          ),
        ],
      ),
    );
  }
}
